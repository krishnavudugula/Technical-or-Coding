class Methods {
    static void greet(){
        System.out.println("Hello, World!");
    }
    static void greet(String name){
        System.out.println("Hiiiii " + name);
    }
    static int add(int a, int b){
        return a+b;
    }
    static double square(int s){
        return s*s;
    }
    static boolean e_or_o(int num){
        return num%2==0;
    }
    static int max_num(int a, int b){
        if(a>b){
            return a;
        }
        else {
            return b;
        }
    }
    static long area_o_rect(int length, int width){
        return length * width;
    } 
    static long factorial(int n) {
        long result = 1;  
        for (int i = 1; i <= n; i++) {
            result *= i;  
        }
        return result;
    }
    
    public static void main(String[] args){
        greet();
        greet("Krishna");
        
        int sum = add(2,3);
        System.out.println(sum);
        
        double sq = square(12);
        System.out.println(sq);
        
        boolean result = e_or_o(12);
        System.out.println(result);
        
        int max = max_num(14,13);
        System.out.println(max);
        
        long area = area_o_rect(5,4);
        System.out.println(area);
        
        long number = factorial(5);
        System.out.println(number);
    }
    
    
    
    
    
    
}
